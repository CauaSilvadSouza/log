Introdução

Neste projeto, desenvolvemos um sistema para consultar e visualizar cotações históricas do dólar PTAX utilizando a API pública do Banco Central do Brasil. O PTAX é a taxa de câmbio oficial usada como referência para operações no mercado brasileiro.

O que é PTAX?

A PTAX (Preço de Taxa de Câmbio) é calculada pelo Banco Central do Brasil com base nas operações de compra e venda de dólares realizadas durante o dia. É amplamente utilizada para:

  • Contratos de câmbio
  • Operações de importação/exportação
  • Referência para investimentos
  • Cálculos contábeis

Arquitetura da Solução

O sistema é composto por:

  1. Cliente API: Conexão com o serviço OData do BCB
  2. Processamento: Tratamento e preenchimento de dados
  3. Visualização: Gráficos interativos com Plotly

Implementação

Classe Principal

import requests
import pandas as pd
import plotly.graph_objects as go
from datetime import datetime
import calendar


class ConsultadorDolarPTAX:
    """Classe para consultar e visualizar cotações do dólar"""
    
    def __init__(self, periodo: str):
        self.periodo = periodo
        self.url_api = "https://olinda.bcb.gov.br/olinda/servico/PTAX/versao/v1/odata/"
        self.dados_processados = None
        
    def _converter_periodo(self):
        """Converte string MMYYYY para objeto datetime"""
        try:
            return datetime.strptime(self.periodo, "%m%Y")
        except ValueError:
            raise ValueError("Período deve estar no formato MMYYYY")

Montagem da URL de Consulta

O método constrói a URL com base no período solicitado:

    def _montar_url_consulta(self, dt_objeto):
        """Monta URL de consulta da API"""
        primeiro_dia = 1
        ultimo_dia = calendar.monthrange(dt_objeto.year, dt_objeto.month)[1]
        
        dt_inicio = f"{dt_objeto.month:02d}-{primeiro_dia:02d}-{dt_objeto.year}"
        dt_fim = f"{dt_objeto.month:02d}-{ultimo_dia:02d}-{dt_objeto.year}"
        
        endpoint = (
            f"{self.url_api}CotacaoDolarPeriodo(dataInicial=@dataInicial,"
            f"dataFinalCotacao=@dataFinalCotacao)?"
            f"@dataInicial='{dt_inicio}'&@dataFinalCotacao='{dt_fim}'&$format=json"
        )
        
        return endpoint, primeiro_dia, ultimo_dia

Busca de Cotações

    def buscar_cotacoes(self):
        """Realiza a busca das cotações na API do BCB"""
        print(f"🔍 Buscando cotações para {self.periodo}...")
        
        dt_referencia = self._converter_periodo()
        url_requisicao, dia_ini, dia_fim = self._montar_url_consulta(dt_referencia)
        
        try:
            resposta = requests.get(url_requisicao, timeout=10)
            resposta.raise_for_status()
            dados_json = resposta.json()
            
            if not dados_json.get('value'):
                print(f"⚠️  Nenhuma cotação encontrada para {self.periodo}")
                return None
            
            # Processar dados da API
            dataframe = pd.DataFrame(dados_json['value'])
            dataframe['dataHoraCotacao'] = pd.to_datetime(dataframe['dataHoraCotacao'])
            dataframe['dia'] = dataframe['dataHoraCotacao'].dt.date
            dataframe = dataframe[['dia', 'cotacaoVenda']].sort_values('dia')
            
            print(f"✅ {len(dataframe)} registros obtidos")
            
            # Preencher todos os dias do mês
            dataframe_completo = self._preencher_dias_faltantes(
                dataframe, dt_referencia, dia_ini, dia_fim
            )
            
            self.dados_processados = dataframe_completo
            return dataframe_completo
            
        except requests.exceptions.Timeout:
            print("❌ Tempo esgotado ao conectar com a API")
            return None
        except requests.exceptions.RequestException as erro:
            print(f"❌ Erro na requisição: {erro}")
            return None

Preenchimento de Dados Faltantes

Como não há cotação em finais de semana e feriados, o método preenche esses dias com o último valor disponível:

    def _preencher_dias_faltantes(self, df_original, dt_ref, dia_inicial, dia_final):
        """Preenche dias sem cotação com o último valor disponível"""
        data_inicio = datetime(dt_ref.year, dt_ref.month, dia_inicial).date()
        data_fim = datetime(dt_ref.year, dt_ref.month, dia_final).date()
        
        # Criar range com todos os dias
        todos_dias = pd.date_range(start=data_inicio, end=data_fim, freq='D').date
        df_todos_dias = pd.DataFrame({'dia': todos_dias})
        
        # Fazer merge e preencher valores faltantes
        df_final = df_todos_dias.merge(df_original, on='dia', how='left')
        df_final['cotacaoVenda'].fillna(method='ffill', inplace=True)
        df_final.rename(columns={'cotacaoVenda': 'valor_cotacao'}, inplace=True)
        
        dias_preenchidos = df_final['valor_cotacao'].notna().sum() - len(df_original)
        if dias_preenchidos > 0:
            print(f"📝 {dias_preenchidos} dias foram preenchidos com valores anteriores")
        
        return df_final

Geração de Gráficos

    def gerar_grafico(self):
        """Gera gráfico interativo das cotações"""
        if self.dados_processados is None:
            print("❌ Não há dados para gerar o gráfico")
            return None
        
        df = self.dados_processados
        dt_ref = self._converter_periodo()
        nome_mes = dt_ref.strftime("%B").upper()
        
        # Criar gráfico com plotly graph_objects
        figura = go.Figure()
        
        figura.add_trace(go.Scatter(
            x=df['dia'],
            y=df['valor_cotacao'],
            mode='lines+markers',
            name='Cotação',
            line=dict(color='#00a86b', width=2.5),
            marker=dict(size=5, color='#006b4e'),
            hovertemplate='<b>Data:</b> %{x}<br><b>Cotação:</b> R$ %{y:.4f}<extra></extra>'
        ))
        
        figura.update_layout(
            title=f'COTAÇÃO PTAX DO DÓLAR - {nome_mes}/{dt_ref.year}',
            xaxis_title='Período',
            yaxis_title='Valor em Reais (R$)',
            plot_bgcolor='white',
            hovermode='x unified',
            height=550
        )
        
        # Salvar gráfico
        nome_arquivo = f"grafico_dolar_ptax_{self.periodo}.html"
        figura.write_html(nome_arquivo, include_plotlyjs='cdn')
        
        print(f"\n✅ Gráfico salvo: {nome_arquivo}")
        
        return nome_arquivo

Uso do Sistema

def executar_consulta(mes_ano: str):
    """Função principal para executar a consulta"""
    try:
        if len(mes_ano) != 6 or not mes_ano.isdigit():
            print("\n❌ Formato incorreto! Utilize MMYYYY (exemplo: 072016)")
            return
        
        print("\n" + "="*75)
        print("SISTEMA DE CONSULTA - COTAÇÃO DÓLAR PTAX (BCB)")
        print("="*75 + "\n")
        
        consultor = ConsultadorDolarPTAX(mes_ano)
        
        # Buscar cotações
        dados = consultor.buscar_cotacoes()
        
        if dados is not None:
            # Gerar gráfico
            consultor.gerar_grafico()
            
            # Exibir estatísticas
            print("\n📊 ESTATÍSTICAS DO PERÍODO:")
            print(f"   • Valor mínimo: R$ {dados['valor_cotacao'].min():.4f}")
            print(f"   • Valor máximo: R$ {dados['valor_cotacao'].max():.4f}")
            print(f"   • Valor médio: R$ {dados['valor_cotacao'].mean():.4f}")
        
        print("\n" + "="*75)
        
    except Exception as erro:
        print(f"\n❌ Erro durante execução: {erro}")


if __name__ == "__main__":
    periodo_consulta = "022019"  # Fevereiro de 2019
    executar_consulta(periodo_consulta)

## Classe Principal

```{python}
class RastreadorOnibusAPI:
    """Sistema de rastreamento de ônibus usando API SPTrans OlhoVivo"""
    
    def __init__(self, chave_api):
        self.chave_api = chave_api
        self.sessao_http = requests.Session()
        self.url_base = "http://api.olhovivo.sptrans.com.br/v2.1"
        self.autenticado = False
        
    def autenticar(self):
        """Autentica na API da SPTrans"""
        endpoint = f"{self.url_base}/Login/Autenticar?token={self.chave_api}"
        
        try:
            resposta = self.sessao_http.post(endpoint, timeout=10)
            
            if resposta.text.lower() == "true":
                print("✅ Conexão estabelecida com sucesso!")
                self.autenticado = True
                return True
            else:
                print("❌ Falha na autenticação - Token inválido")
                return False
                
        except Exception as erro:
            print(f"❌ Erro na autenticação: {erro}")
            return False
    
    def pesquisar_linhas(self, termo):
        """Busca linhas de ônibus por termo"""
        if not self.autenticado:
            print("❌ Autentique primeiro!")
            return []
        
        endpoint = f"{self.url_base}/Linha/Buscar?termosBusca={termo}"
        resposta = self.sessao_http.get(endpoint)
        return resposta.json()
    
    def obter_paradas_linha(self, cod_linha):
        """Obtém todas as paradas de uma linha específica"""
        if not self.autenticado:
            print("❌ Autentique primeiro!")
            return []
        
        endpoint = f"{self.url_base}/Parada/BuscarParadasPorLinha?codigoLinha={cod_linha}"
        resposta = self.sessao_http.get(endpoint)
        return resposta.json()
    
    def obter_posicoes_veiculos(self, cod_linha):
        """Obtém posições em tempo real dos veículos"""
        if not self.autenticado:
            print("❌ Autentique primeiro!")
            return {}
        
        endpoint = f"{self.url_base}/Posicao/Linha?codigoLinha={cod_linha}"
        resposta = self.sessao_http.get(endpoint)
        return resposta.json()
    
    def gerar_mapa_interativo(self, cod_linha, identificador_linha):
        """Gera mapa interativo com paradas e veículos"""
        
        print(f"\n🗺️  Gerando mapa para linha {identificador_linha}...")
        
        # Obter paradas
        lista_paradas = self.obter_paradas_linha(cod_linha)
        qtd_paradas = len(lista_paradas)
        print(f"   📍 Paradas encontradas: {qtd_paradas}")
        
        # Obter posições dos veículos
        dados_veiculos = self.obter_posicoes_veiculos(cod_linha)
        lista_veiculos = dados_veiculos.get('vs', [])
        horario_ref = dados_veiculos.get('hr', 'Indisponível')
        qtd_veiculos = len(lista_veiculos)
        print(f"   🚌 Veículos em circulação: {qtd_veiculos}")
        print(f"   🕐 Timestamp: {horario_ref}")
        
        # Calcular centro do mapa
        if lista_paradas:
            lat_media = sum(parada['py'] for parada in lista_paradas) / qtd_paradas
            lon_media = sum(parada['px'] for parada in lista_paradas) / qtd_paradas
        else:
            # Coordenadas padrão (Centro de São Paulo)
            lat_media, lon_media = -23.5505, -46.6333
        
        # Criar mapa base
        mapa_folium = folium.Map(
            location=[lat_media, lon_media],
            zoom_start=14,
            tiles='OpenStreetMap',
            control_scale=True
        )
        
        # Adicionar paradas ao mapa
        for parada in lista_paradas:
            info_popup = f"""
            <div style="font-family: Arial; width: 250px;">
                <h4 style="margin: 0 0 10px 0; color: #1976D2;">
                    📍 {parada['np']}
                </h4>
                <p style="margin: 5px 0;"><b>Local:</b> {parada['ed']}</p>
                <p style="margin: 5px 0;"><b>Código:</b> {parada['cp']}</p>
            </div>
            """
            
            CircleMarker(
                location=[parada['py'], parada['px']],
                radius=6,
                popup=folium.Popup(info_popup, max_width=280),
                tooltip=f"Parada: {parada['np']}",
                color='#1976D2',
                fill=True,
                fillColor='#2196F3',
                fillOpacity=0.7
            ).add_to(mapa_folium)
        
        # Adicionar veículos ao mapa
        for veiculo in lista_veiculos:
            eh_acessivel = "✓ Sim" if veiculo.get('a', False) else "✗ Não"
            horario_atualizacao = veiculo.get('ta', 'N/A')
            prefixo = veiculo['p']
            
            info_veiculo = f"""
            <div style="font-family: Arial; width: 250px;">
                <h4 style="margin: 0 0 10px 0; color: #D32F2F;">
                    🚍 Veículo #{prefixo}
                </h4>
                <p style="margin: 5px 0;"><b>Acessibilidade:</b> {eh_acessivel}</p>
                <p style="margin: 5px 0;"><b>Atualizado em:</b> {horario_atualizacao}</p>
            </div>
            """
            
            Marker(
                location=[veiculo['py'], veiculo['px']],
                popup=folium.Popup(info_veiculo, max_width=280),
                tooltip=f"Ônibus {prefixo}",
                icon=Icon(color='red', icon='bus', prefix='fa')
            ).add_to(mapa_folium)
        
        print(f"✅ Mapa gerado com sucesso!")
        
        return mapa_folium, {
            'paradas': qtd_paradas,
            'veiculos': qtd_veiculos,
            'horario': horario_ref
        }
```
